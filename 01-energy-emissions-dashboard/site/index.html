import {areaY, barX, barY, geo, image, rect, text, plot} from "./plot.js";
// Quarto OJS can't parse `import * as`, so rebuild the namespace by hand.
// Any new Plot mark used below must be added to the import above.
Plot = ({areaY, barX, barY, geo, image, rect, text, plot});
gen = fetch("data/generation.json").then(r => r.json());
factors = fetch("data/factors.json").then(r => r.json());
nonEl = fetch("data/nonElectric.json").then(r => r.json());
scenarios = fetch("data/scenarios.json").then(r => r.json());
paris = fetch("data/paris.json").then(r => r.json());
states = fetch("data/states.geojson").then(r => r.json());
forestUrl = "data/forest.png";
forestBounds = fetch("data/forestBounds.json").then(r => r.json());
typeOrder = ["Coal", "Gas", "Wind", "Solar (incl. Rooftop PV)", "Other", "Hydro & Storage"];
palette = (() => ({"Coal": "#3b3b3b", "Gas": "#9aa0a6", "Wind": "#7ec8a9", "Solar (incl. Rooftop PV)": "#ffd54f", "Other": "#8b5a2b", "Hydro & Storage": "#4dabf7"}))();
factorFor = (t) => factors.find(f => f.Type === t)?.mt_per_twh ?? 0;
viewof scen = Inputs.select(scenarios.map(s => s.label), {
  value: scenarios[1].label,
  label: "Electrification scenario"
});
scenario = scenarios.find(s => s.label === scen);
demand2050 = scenario.demand_2050_twh;
viewof coal = Inputs.number({value: 5, min: 0, max: 800, step: 1, label: "Coal, 2050 (TWh)"});
viewof gas = Inputs.number({value: 35, min: 0, max: 800, step: 1, label: "Gas, 2050 (TWh)"});
viewof wind = Inputs.number({value: 205, min: 0, max: 800, step: 1, label: "Wind, 2050 (TWh)"});
viewof solar = Inputs.number({value: 180, min: 0, max: 800, step: 1, label: "Solar incl. rooftop, 2050 (TWh)"});
viewof other = Inputs.number({value: 15, min: 0, max: 800, step: 1, label: "Other, 2050 (TWh)"});
viewof hydro = Inputs.number({value: 60, min: 0, max: 800, step: 1, label: "Hydro & storage, 2050 (TWh)"});
viewof submit = Inputs.button("Update", {value: 0, reduce: (x) => x + 1});
mix = (() => ({Coal: coal, Gas: gas, Wind: wind, "Solar (incl. Rooftop PV)": solar, Other: other, "Hydro & Storage": hydro}))();
mixTotal = typeOrder.reduce((s, t) => s + mix[t], 0);
mixValid = Math.abs(mixTotal - demand2050) < 0.05;
mutable lastApplied = (() => ({Coal: 5, Gas: 35, Wind: 205, "Solar (incl. Rooftop PV)": 180, Other: 15, "Hydro & Storage": 60}))();
{
  submit;
  const els = [viewof coal, viewof gas, viewof wind, viewof solar, viewof other, viewof hydro];
  const m = (() => ({Coal: +els[0].value, Gas: +els[1].value, Wind: +els[2].value, "Solar (incl. Rooftop PV)": +els[3].value, Other: +els[4].value, "Hydro & Storage": +els[5].value}))();
  const t = m.Coal + m.Gas + m.Wind + m["Solar (incl. Rooftop PV)"] + m.Other + m["Hydro & Storage"];
  const target = scenarios.find(s => s.label === viewof scen.value)?.demand_2050_twh ?? demand2050;
  if (Math.abs(t - target) < 0.05) lastApplied.value = m;
}
appliedMix = lastApplied;
statusBanner = mixValid ? html`<div class="ok">Total = ${mixTotal.toFixed(1)} TWh — matches the ${scenario.label} demand. Mix applied on Update.</div>` : html`<div class="warn">Total is ${mixTotal.toFixed(1)} TWh; the scenario target is ${demand2050} TWh. Update is blocked until they match (or use Rescale).</div>`;
viewof rescale = Inputs.button("Rescale to target", {value: 0, reduce: (x) => x + 1});
{
  rescale;
  const els = [viewof coal, viewof gas, viewof wind, viewof solar, viewof other, viewof hydro];
  const vals = els.map(e => +e.value);
  const total = vals.reduce((s, v) => s + v, 0);
  const target = scenarios.find(s => s.label === viewof scen.value)?.demand_2050_twh ?? demand2050;
  if (total > 0) {
    const f = target / total;
    const out = vals.slice(0, 5).map(v => +(v * f).toFixed(1));
    out.push(+(target - out.reduce((s, v) => s + v, 0)).toFixed(2));
    els.forEach((e, i) => {
      e.value = out[i];
      e.dispatchEvent(new Event("input", {bubbles: true}));
    });
  }
}
viewof seqRate = Inputs.range([1, 30], {value: 10, step: 1, label: "Sequestration rate (t CO2 / ha / yr)"});
viewof offsetCeiling = Inputs.range([0, 300], {value: 100, step: 5, label: "Offset ceiling (Mt CO2 / yr)"});
genPath = gen.data.map(d => d.year === 2050 ? {...d, twh: appliedMix[d.Type]} : {...d});
elecByYear = gen.years.map(y => ({year: y, mt: genPath.filter(d => d.year === y).reduce((s, d) => s + d.twh * factorFor(d.Type), 0)}));
nonElPath = nonEl.map(d => d.year === 2050 ? {...d, mt: scenario.non_electric_2050_mt} : {...d});
stackYears = [1990, 2005, 2020, 2025, 2030, 2035, 2040, 2050];
stackData = stackYears.flatMap(y => { const e = elecByYear.find(d => d.year === y)?.mt ?? 0; const n = nonElPath.find(d => d.year === y)?.mt ?? 0; return [{year: y, source: "Electricity", mt: e}, {year: y, source: "Non-electricity", mt: n}]; });
total2050 = (elecByYear.find(d => d.year === 2050)?.mt ?? 0) + (nonElPath.find(d => d.year === 2050)?.mt ?? 0);
onTrack = total2050 <= offsetCeiling;
forestHa = 133564800;
tasHa = 6840000;
plantingHa = total2050 * 1e6 / seqRate;
forestSideM = Math.sqrt(forestHa * 10000);
plantingSideM = Math.sqrt(plantingHa * 10000);
pctOfForest = 100 * plantingHa / forestHa;
timesTas = plantingHa / tasHa;
mapMarks = [
  Plot.geo(states, {fill: "#f3ead7", stroke: "#6b6350", strokeOpacity: 0.85}),
  Plot.rect([{x1: -2250000, y1: -5000000, x2: -2250000 + forestSideM, y2: -5000000 + forestSideM}], {x1: "x1", x2: "x2", y1: "y1", y2: "y2", stroke: "#555", strokeWidth: 2.5, fill: "none"}),
  Plot.rect([{x1: -1050000, y1: -4650000, x2: -1050000 + plantingSideM, y2: -4650000 + plantingSideM}], {x1: "x1", x2: "x2", y1: "y1", y2: "y2", fill: "#1b7a2b", fillOpacity: 0.7, stroke: "#0f4d1c", strokeWidth: 1.5}),
  Plot.text([{x: -1672000, y: -4915000, t: "All existing forest"}], {x: "x", y: "y", text: "t", fontSize: 14, fill: "#333"}),
  Plot.text([{x: -1050000, y: -4585000, t: "New planting"}], {x: "x", y: "y", text: "t", fontSize: 14, fill: "#0f4d1c"})
].filter(Boolean);
// The image mark's width/height channels are pixel units (unscaled), so convert
// the forest extent from EPSG:3577 metres to pixels using this chart's frame.
// The frame is kept equal-aspect (see mapChart) so x and y pixels share a scale.
forestImage = (width, height) => forestBounds ? Plot.image([{
  src: forestUrl,
  x: (forestBounds.xmin + forestBounds.xmax) / 2,
  y: (forestBounds.ymin + forestBounds.ymax) / 2,
  w: (forestBounds.xmax - forestBounds.xmin) * (width - 8) / 5000000,
  h: (forestBounds.ymax - forestBounds.ymin) * (width - 8) / 5000000
}], {x: "x", y: "y", width: "w", height: "h", src: "src", opacity: 0.9}) : null;
mapCaptionNote = html`New planting needed: <b>${(plantingHa / 1e6).toFixed(1)} Mha</b> — ${pctOfForest.toFixed(1)}% of Australia's existing forest (≈ ${timesTas.toFixed(1)} × Tasmania), at ${seqRate} t CO2/ha/yr. Grey square: all existing forest (133.6 Mha, side ${Math.round(forestSideM / 1000)} km). Green square: new planting (side ${Math.round(plantingSideM / 1000)} km). Equal-area projection (EPSG:3577).`;
function genChart(width) {
  return {
    width,
    height: 380,
    marginLeft: 64,
    marginBottom: 40,
    y: {label: "TWh", grid: true},
    x: {label: null, domain: [1990, 2050], ticks: [1990, 2000, 2010, 2020, 2030, 2040, 2050], tickFormat: "d"},
    color: {domain: typeOrder, range: typeOrder.map(t => palette[t]), legend: true},
    marks: [Plot.areaY(genPath, {x: "year", y: "twh", fill: "Type", order: typeOrder, curve: "linear"})]
  };
}
function stackChart(width) {
  return {
    width,
    height: 320,
    marginLeft: 64,
    y: {label: "Mt CO2-e", grid: true},
    x: {type: "band", label: null, tickFormat: "d"},
    color: {domain: ["Electricity", "Non-electricity"], range: ["#d64545", "#7d8590"], legend: true},
    marks: [Plot.barY(stackData, {x: "year", y: "mt", fill: "source", order: ["Non-electricity", "Electricity"]})]
  };
}
function parisChart(width) {
  return {
    width,
    height: 200,
    marginLeft: 110,
    x: {label: "Mt CO2-e", grid: true, domain: [0, 700]},
    y: {label: null},
    marks: [Plot.barX(paris, {y: "label", x: "mt", fill: d => d.mt === 0 ? "#0b4d1c" : d.mt < 400 ? "#0b4d1c" : "#333"}), Plot.text(paris, {y: "label", x: "mt", text: d => d.mt === 0 ? "ZERO" : `${d.mt} Mt`, dx: 10, textAnchor: "start"})]
  };
}
function mapChart(width) {
  // Equal-aspect frame: data extent is 5,000,000 x 4,550,000 m, so the pixel
  // scale is the same on both axes and km squares stay true squares.
  const height = (width - 8) * 4550000 / 5000000 + 8;
  return {
    width,
    height,
    margin: 4,
    x: {domain: [-2300000, 2700000], axis: null},
    y: {domain: [-5350000, -800000], axis: null},
    aspectRatio: false,
    marks: [
      Plot.rect([{x1: -2300000, y1: -5350000, x2: 2700000, y2: -800000}], {x1: "x1", x2: "x2", y1: "y1", y2: "y2", fill: "#eef4fa", stroke: "none"}),
      forestImage(width, height),
      ...mapMarks
    ]
  };
}

The path to 2050

width > 0 ? Plot.plot(genChart(width)) : null

G_{s,2050} = x_s \qquad \sum_s x_s = D_{2050} \qquad \text{Rescale: } x_s \leftarrow x_s \cdot \frac{D_{2050}}{\sum_k x_k}

G_{s,t} generation from source s in year t (TWh) · x_s your 2050 input for source s · D_{2050} 2050 demand under the chosen electrification scenario. Years before 2050 follow the historical record and the AEMO Step Change path.

E^{\text{el}}_t electricity emissions in year t (Mt CO2-e) · f_s emission factor of source s (Mt CO2/TWh: coal 1.01, gas 0.49, other 0.41, renewables and storage 0) · E^{\text{ne}}_t non-electricity emissions · O offset ceiling (Mt CO2/yr). Targets: E^{*}_{2030} = (1 - 0.43)\,E_{2005} and net zero in 2050.

Assumptions & sources

html`
<ul class="assumptions">
  <li><b>Generation history & path.</b> 1990–2025 history and the 2030–2040 path from Australian Energy Statistics 2024 (Table O) and the AEMO 2024 ISP step-change scenario. The 2050 column is your mix; intermediate years keep the AEMO path unchanged.</li>
  <li><b>Emission factors (Mt CO2/TWh).</b> Coal 1.01, gas 0.49, other 0.41, renewables & storage 0 — fixed per source.</li>
  <li><b>Non-electricity emissions.</b> DCCEEW Australia's Emissions Projections 2024 (total minus electricity), with the 2050 endpoint following the electrification scenario.</li>
  <li><b>Electrification scenarios.</b> Simplified mapping: low / central (AEMO Step Change) / high electrification set the 2050 demand total and the 2050 non-electricity endpoint; intermediate years are unchanged.</li>
  <li><b>Paris targets.</b> 2005 baseline 613.42 Mt (DCCEEW 2024); 2030 target −43% vs 2005 = 349.65 Mt (Climate Change Act 2022); 2050 net zero.</li>
  <li><b>Offset ceiling.</b> Illustrative, adjustable: how much residual CO2-e could realistically be offset each year (default 100 Mt/yr).</li>
  <li><b>Sequestration.</b> Adjustable, default 10 t CO2/ha/yr; real rates vary strongly with rainfall and management.</li>
  <li><b>Forest map.</b> ABARES Forests of Australia 2023 (CC BY 4.0): 133.6 Mha of forest, aggregated to 2.5 km cells and shaded from pale to deep green by the share of each cell that is forested; state outlines from ozmaps (ABS). Tasmania = 6.84 Mha.</li>
  <li><b>This is a simplified model</b> for exploring trade-offs, not a forecast.</li>
</ul>`
